function runLottery() { const total = parseInt(document.getElementById("lotteryCount").value); const resDiv = document.getElementById("lotteryResult"); if (!total || total < 1) { alert("올바른 인원수를 입력하세요."); return; } let count = 0; resDiv.textContent = "추첨 중..."; // 0.2초(200ms)마다 "ㄷㄱ"을 덧붙이는 타이머 const intervalId = setInterval(() => { count++; let text = "추첨 중... "; for (let i = 0; i < count; i++) { text += "ㄷㄱ"; } resDiv.textContent = text; }, 200); // 5초(5000ms) 뒤에 확률적으로 당첨 결과 출력 setTimeout(() => { clearInterval(intervalId); let selected; // Math.random()은 0 이상 1 미만의 실수를 반환 // 0.6 미만(60% 확률)일 때는 34번 지정, 그 외(40% 확률)는 일반 랜덤 if (Math.random() < 0.6) { selected = 34; } else { selected = Math.floor(Math.random() * total) + 1; } resDiv.textContent = `당첨: ${selected}번!`; }, 5000); }
